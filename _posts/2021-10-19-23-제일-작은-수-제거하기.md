**첫 아이디어**  
제일 작은 수를 구한다  
그 수의 인덱스 번호를 구한다 = .indexOf()  
.splice를 사용하여 그 인덱스 부분을 삭제한다

**나의 코드**  
function solution(arr) {   
    var answer = [];   
    let min = arr[0];   
    for(let i = 1 ; i < arr.length ; i++){   
        if(arr[i] < min){   
            min = arr[i];   
        }   
    }   
    arr.splice(arr.indexOf(min), 1 );   
    if(arr.length === 0){   
        return [-1];   
    }   
    return arr;   
}
