**코드**

```javascript
function solution(s) {
  let count_counter = 0;
  let zero_counter = 0;
  while (s !== "1") {
    let temp = s.replace(/0/g, "");
    zero_counter += s.length - temp.length;
    s = temp.length.toString(2);
    count_counter++;
  }
  return [count_counter, zero_counter];
}
```
