**아이디어**

정규표현식 replace 쓰면되겠다

**코드**

```javascript
function solution(s) {
  const result = s
    .replace(/zero/g, 0)
    .replace(/one/g, 1)
    .replace(/two/g, 2)
    .replace(/three/g, 3)
    .replace(/four/g, 4)
    .replace(/five/g, 5)
    .replace(/six/g, 6)
    .replace(/seven/g, 7)
    .replace(/eight/g, 8)
    .replace(/nine/g, 9);
  return Number(result);
}
```

**다른 방식의 코드 (split과 join활용)**

```javascript
function solution(s) {
    let numbers = ["zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine"];
    var answer = s;
    for(let i=0; i< numbers.length; i++) {
        let arr = answer.split(numbers[i]); //'abcd'.split('b') === ['a', 'cd']
        answer = arr.join(i);  // ['a', 'cd'].join(10) === 'a10cd'
    }
    return Number(answer);
}
```
