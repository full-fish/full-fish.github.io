```javascript
.sort(({ properties: { name: a = 0 } } = {}, { properties: { name: b = 0 } } = {}) => (a.match(/\d+/) ?? [0])[0] - (b.match(/\d+/) ?? [0])[0])
.sort(({ properties: { name: a = '가' } } = {}, { properties: { name: b = '가' } } = {}) =>
  (a.match(/[가-힣]/) ?? ['가'])[0] > (b.match(/[가-힣]/) ?? ['가'])[0]
    ? 1
    : (a.match(/[가-힣]/) ?? ['가'])[0] < (b.match(/[가-힣]/) ?? ['가'])[0]
    ? -1
    : 0
)
```
