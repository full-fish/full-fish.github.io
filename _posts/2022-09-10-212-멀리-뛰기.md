**코드**

```javascript
function solution(n) {
  let fiboArr = [1, 1];
  for (let i = 2; i <= n; i++) {
    fiboArr.push((fiboArr[i - 1] + fiboArr[i - 2]) % 1234567);
  }
  return fiboArr[n] % 1234567;
}
```
