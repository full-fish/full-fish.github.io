**첫 아이디어**

나누어 떨어지는것을 push로 집어 넣고

answer.sort(function(a, b)  {   
    return a - b;   
})

새로 배운 이 함수로 오름차순으로 만든다

return b-a면 내림차순

**나의 코드**  
function solution(arr, divisor) {   
    var answer = [];   
       
    for(let i = 0 ;i < arr.length ; i++){   
        if(arr[i] % divisor === 0){   
        answer.push(arr[i]);    
        }   
    }   
  
    answer.sort(function(a, b)  {   
        return a - b;   
    });   
    if(answer.length === 0) {   
        answer.push(-1);   
    }   
        return answer;   
       
}
