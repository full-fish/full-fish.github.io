**초기 코딩**

function solution(x) {   
    var answer = true;   
    return answer;   
}

**첫 아이디어**

숫자 x를 문자열로 바꾸고 split으로 배열화 시킨다

배열의 element를 숫자로 바꾸고 더하고 푼다

**나의 코드**  
function solution(x) {   
    var answer = true;   
    let sum = 0,  arrx = [];   
    arrx=String(x).split('');   
    for(let i = 0 ; i < arrx.length ; i++){   
    sum += Number(arrx[i]);   
    }   
    if(x % sum !==0) {   
        answer = false;   
    }   
    return answer;   
}
