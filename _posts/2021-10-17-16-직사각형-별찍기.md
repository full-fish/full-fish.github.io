**나의 코드**  
process.stdin.setEncoding('utf8');   
process.stdin.on('data', data => {   
    const n = data.split(" ");   
    const a = Number(n[0]), b = Number(n[1]);   
    let stars = '';   
  
    for(let i=0; i<b; i++) {   
        for (let j=0; j<a; j++) {   
            stars += '\*'   
        }   
        stars += '\n'   
    }   
    console.log(stars)   
});
