**첫 아이디어**그냥 더하고 나누자

**나의 코드**function solution(arr) {   
    var answer = 0;   
    for(let i = 0 ; i < arr.length ; i++){   
        answer += arr[i];   
    }   
    answer = answer/arr.length;   
    return answer;   
}
