**코드**

```javascript
function solution(s) {
  return s
    .split("")
    .sort(function (a, b) {
      if (a < b) return 1;
      if (a > b) return -1;
      if (a === b) return 0;
    })
    .join("");
}

// 다른 방법 
// 정렬하고 뒤집기

function solution(s) {
  return s.split("").sort().reverse().join("");
}
```
